cmake_minimum_required(VERSION 3.19)
project(H2SweepBench LANGUAGES CXX)
set(SPANN_UPSTREAM_BUILD "" CACHE PATH "Existing clean pinned upstream Release build")
set(recipe "${SPANN_UPSTREAM_BUILD}/AnnService/CMakeFiles/ssdserving.dir")
file(STRINGS "${SPANN_UPSTREAM_BUILD}/CMakeCache.txt" home REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" native_source "${home}")
find_package(Git REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${native_source}" rev-parse HEAD
    OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${native_source}" status --porcelain --untracked-files=no
    OUTPUT_VARIABLE dirty COMMAND_ERROR_IS_FATAL ANY)
if(NOT revision STREQUAL "2ac3ebcab562bc81cdb8c7c98b35ea72f2703c3b" OR NOT dirty STREQUAL "")
    message(FATAL_ERROR "Require clean pinned upstream 2ac3ebc")
endif()
foreach(name DEFINES INCLUDES FLAGS)
    file(STRINGS "${recipe}/flags.make" line REGEX "^CXX_${name} = ")
    string(REGEX REPLACE "^CXX_${name} = " "" native_${name} "${line}")
endforeach()
file(STRINGS "${recipe}/flags.make" line REGEX "^# compile CXX with ")
string(REPLACE "# compile CXX with " "" native_compiler "${line}")
get_filename_component(expected "${native_compiler}" REALPATH)
get_filename_component(actual "${CMAKE_CXX_COMPILER}" REALPATH)
if(NOT expected STREQUAL actual OR NOT native_FLAGS MATCHES "-O3" OR
   NOT native_FLAGS MATCHES "-std=c\\+\\+17" OR NOT native_FLAGS MATCHES "-fopenmp")
    message(FATAL_ERROR "Require native compiler and C++17 OpenMP Release recipe")
endif()
set(CMAKE_CXX_FLAGS "${native_FLAGS}")
set(CMAKE_CXX_FLAGS_RELEASE "")
separate_arguments(definitions UNIX_COMMAND "${native_DEFINES}")
separate_arguments(includes UNIX_COMMAND "${native_INCLUDES}")
list(TRANSFORM definitions REPLACE "^-D" "")
list(TRANSFORM includes REPLACE "^-I" "")
file(READ "${recipe}/link.txt" link_text)
separate_arguments(link UNIX_COMMAND "${link_text}")
list(FIND link "-o" position)
if(position LESS 0)
    message(FATAL_ERROR "Missing native link output")
endif()
math(EXPR position "${position} + 2")
list(SUBLIST link ${position} -1 libraries)
if(NOT "${libraries}" MATCHES "/libSPTAGLibStatic\\.a(;|$)")
    message(FATAL_ERROR "Need native static archive")
endif()
set(suffix "")
foreach(library IN LISTS libraries)
    string(APPEND suffix " \"${library}\"")
endforeach()
set(CMAKE_CXX_LINK_EXECUTABLE
    "<CMAKE_CXX_COMPILER> <FLAGS> <CMAKE_CXX_LINK_FLAGS> <LINK_FLAGS> <OBJECTS> -o <TARGET>${suffix}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/instrument_bkt.py"
    "${native_source}" "${CMAKE_CURRENT_BINARY_DIR}/isolated"
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(h2sweepbench ../H2SweepBench.cpp "${CMAKE_CURRENT_BINARY_DIR}/isolated/BKTIndex.cpp")
target_compile_definitions(h2sweepbench PRIVATE H2_CALLBACK_COUNTER ${definitions})
target_include_directories(h2sweepbench BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/isolated" ${includes})
add_executable(h2sweepbench_ordinary ../H2SweepBench.cpp)
target_compile_definitions(h2sweepbench_ordinary PRIVATE ${definitions})
target_include_directories(h2sweepbench_ordinary PRIVATE ${includes})
foreach(target h2sweepbench h2sweepbench_ordinary)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
endforeach()
enable_testing()
add_test(NAME h2_sweep_fixture COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_h2_sweep.py"
    "$<TARGET_FILE:h2sweepbench>" "$<TARGET_FILE:h2sweepbench_ordinary>")
set_tests_properties(h2_sweep_fixture PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 180)
