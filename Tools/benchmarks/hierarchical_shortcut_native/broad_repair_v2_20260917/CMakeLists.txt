cmake_minimum_required(VERSION 3.16)
project(BroadRepair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(OpenMP REQUIRED)
find_package(Boost REQUIRED COMPONENTS system thread filesystem regex serialization wserialization)
find_package(TBB REQUIRED)
add_executable(matched-repair MatchedBench.cpp "${SPANN_ROOT}/Wrappers/src/CoreInterface.cpp")
target_compile_definitions(matched-repair PRIVATE MATCHED_CURRENT MATCHED_REPAIR)
add_executable(matched-bad MatchedBench.cpp "${BAD_ROOT}/../build/AnnService/CMakeFiles/spannaclbench.dir/__/Wrappers/src/CoreInterface.cpp.o")
target_compile_definitions(matched-bad PRIVATE MATCHED_CURRENT TBB NUMA)
target_compile_options(matched-bad PRIVATE -fno-omit-frame-pointer)
target_include_directories(matched-bad PRIVATE "${BAD_ROOT}/AnnService" "${BAD_ROOT}/Wrappers")
target_link_libraries(matched-bad PRIVATE "${BAD_ROOT}/Release/libSPTAGLibStatic.a"
    "${BAD_ROOT}/Release/libDistanceUtils.a" "${BAD_ROOT}/Release/libRaBitQ2Lib.a"
    "${BAD_ROOT}/Release/libzstd.a" ${Boost_LIBRARIES} TBB::tbb OpenMP::OpenMP_CXX numa dl rt)
add_executable(native-tests NativeTests.cpp)
foreach(target matched-repair native-tests)
    target_compile_definitions(${target} PRIVATE TBB NUMA)
    target_compile_options(${target} PRIVATE -fno-omit-frame-pointer)
    target_include_directories(${target} PRIVATE "${SPANN_ROOT}/AnnService" "${SPANN_ROOT}/Wrappers")
    target_link_libraries(${target} PRIVATE "${SPANN_ROOT}/Release/libSPTAGLibStatic.a"
        "${SPANN_ROOT}/Release/libDistanceUtils.a" "${SPANN_ROOT}/Release/libRaBitQ2Lib.a"
        "${SPANN_ROOT}/Release/libzstd.a" ${Boost_LIBRARIES} TBB::tbb OpenMP::OpenMP_CXX numa dl rt)
endforeach()
add_test(NAME native-fixtures COMMAND native-tests)
add_executable(allocation-fixture AllocationFixture.cpp)
target_include_directories(allocation-fixture PRIVATE "${SPANN_ROOT}/AnnService")
add_test(NAME allocation-free-expansion COMMAND allocation-fixture)
