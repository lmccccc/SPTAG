cmake_minimum_required(VERSION 3.16)
project(HeadQualityBench LANGUAGES CXX)

# Independent offline tool; consume, never rebuild or modify, the native build.
# cmake -S SPTAG/Tools/benchmarks/head_quality_native -B <prefix>/head-quality-build \
#   -DSPANN_UPSTREAM_BUILD=<prefix>/build -DCMAKE_BUILD_TYPE=Release
# cmake --build <prefix>/head-quality-build --parallel 1
# ctest --test-dir <prefix>/head-quality-build --output-on-failure
set(SPANN_UPSTREAM_BUILD "" CACHE PATH "Existing pinned Microsoft SPANN build")
set(native_target "${SPANN_UPSTREAM_BUILD}/AnnService/CMakeFiles/ssdserving.dir")
foreach(file flags.make link.txt)
    if(NOT EXISTS "${native_target}/${file}")
        message(FATAL_ERROR "Missing native recipe: ${native_target}/${file}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${native_target}/${file}")
endforeach()
file(STRINGS "${SPANN_UPSTREAM_BUILD}/CMakeCache.txt" native_home REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" native_source "${native_home}")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${native_source}" rev-parse HEAD
    OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
if(NOT result EQUAL 0 OR NOT revision STREQUAL "2ac3ebcab562bc81cdb8c7c98b35ea72f2703c3b")
    message(FATAL_ERROR "Expected pinned native revision 2ac3ebc")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${native_source}" status --porcelain --untracked-files=no
    OUTPUT_VARIABLE dirty RESULT_VARIABLE result)
if(NOT result EQUAL 0 OR NOT dirty STREQUAL "")
    message(FATAL_ERROR "Native source must remain clean")
endif()
foreach(name DEFINES INCLUDES FLAGS)
    file(STRINGS "${native_target}/flags.make" line REGEX "^CXX_${name} = ")
    string(REGEX REPLACE "^CXX_${name} = " "" native_${name} "${line}")
endforeach()
file(STRINGS "${native_target}/flags.make" compiler_line REGEX "^# compile CXX with ")
string(REPLACE "# compile CXX with " "" native_compiler "${compiler_line}")
get_filename_component(native_compiler_real "${native_compiler}" REALPATH)
get_filename_component(compiler_real "${CMAKE_CXX_COMPILER}" REALPATH)
if(NOT compiler_real STREQUAL native_compiler_real)
    message(FATAL_ERROR "Use the native compiler: ${native_compiler}")
endif()
if(NOT native_FLAGS MATCHES "-O3" OR NOT native_FLAGS MATCHES "-DNDEBUG" OR
   NOT native_FLAGS MATCHES "-std=c\\+\\+17" OR NOT native_FLAGS MATCHES "-fopenmp")
    message(FATAL_ERROR "Expected native C++17 OpenMP Release flags")
endif()
if(CMAKE_BUILD_TYPE AND NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Only Release is supported")
endif()
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_CXX_FLAGS "${native_FLAGS}")
set(CMAKE_CXX_FLAGS_RELEASE "")
separate_arguments(native_definitions UNIX_COMMAND "${native_DEFINES}")
separate_arguments(native_includes UNIX_COMMAND "${native_INCLUDES}")
list(TRANSFORM native_definitions REPLACE "^-D" "")
list(TRANSFORM native_includes REPLACE "^-I" "")

file(READ "${native_target}/link.txt" native_link_text)
separate_arguments(native_link UNIX_COMMAND "${native_link_text}")
list(FIND native_link "-o" position)
if(position LESS 0)
    message(FATAL_ERROR "Missing native link output")
endif()
math(EXPR position "${position} + 1")
list(GET native_link ${position} native_executable)
get_filename_component(native_bin "${native_executable}" DIRECTORY)
math(EXPR position "${position} + 1")
list(SUBLIST native_link ${position} -1 native_libraries)
if(NOT "${native_libraries}" MATCHES "/libSPTAGLibStatic\\.a(;|$)")
    message(FATAL_ERROR "Native static SPTAG library is required")
endif()
# Preserve the generated ordered suffix, including repeated -ltbb.
set(native_link_suffix "")
set(native_link_files "")
foreach(library IN LISTS native_libraries)
    string(REPLACE "\\" "\\\\" quoted "${library}")
    string(REPLACE "\"" "\\\"" quoted "${quoted}")
    string(APPEND native_link_suffix " \"${quoted}\"")
    if(IS_ABSOLUTE "${library}")
        list(APPEND native_link_files "${library}")
    endif()
endforeach()
set(CMAKE_CXX_LINK_EXECUTABLE
    "<CMAKE_CXX_COMPILER> <FLAGS> <CMAKE_CXX_LINK_FLAGS> <LINK_FLAGS> <OBJECTS> -o <TARGET>${native_link_suffix}")
add_executable(headqualitybench ../HeadQualityBench.cpp)
add_executable(headroutingbench ../HeadQualityBench.cpp)
target_compile_definitions(headroutingbench PRIVATE ${native_definitions})
target_include_directories(headroutingbench PRIVATE ${native_includes})
set_target_properties(headroutingbench PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin" LINK_DEPENDS "${native_link_files}")
target_compile_definitions(headqualitybench PRIVATE ${native_definitions})
target_include_directories(headqualitybench PRIVATE ${native_includes})
set_target_properties(headqualitybench PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${native_bin}" LINK_DEPENDS "${native_link_files}")

enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME head_routing_fixtures COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_head_routing.py" "$<TARGET_FILE:headroutingbench>")
add_test(NAME head_quality_fixtures COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_head_quality.py" "$<TARGET_FILE:headqualitybench>")
set_tests_properties(head_quality_fixtures PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
message(STATUS "Offline analyzer only: ${native_bin}/headqualitybench")
